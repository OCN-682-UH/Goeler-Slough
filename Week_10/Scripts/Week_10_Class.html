Intro into {stringr}

What is a string?

A string and a character are the same thing. You can tell something is a string by the presence of quotations.

words <- "This is a string"
words
[1] "This is a string"
words_vector <- c("Apples", "Bananas", "Oranges")
words_vector
[1] "Apples"  "Bananas" "Oranges"

2. Whitespace

Have some words with extra white spaces:

badtreatments <- c("High", " High", "High ", "Low", "Low")
badtreatments
[1] "High"  " High" "High " "Low"   "Low"

Trimming white space:

str_trim(badtreatments) #this removes white space from both sides
[1] "High" "High" "High" "Low"  "Low" 
str_trim(badtreatments, side = "left") #this removes from only the left side
[1] "High"  "High"  "High " "Low"   "Low"

Adding white space:

str_pad(badtreatments, 5, side = "right") #add a white space to the right side after the 5th character (each will be 5 characters for the final)
[1] "High " " High" "High " "Low  " "Low  "

Adding a character instead of white space:

str_pad(badtreatments, 5, side = "right", pad = "1") # add a 1 to the right side after the 5th character
[1] "High1" " High" "High " "Low11" "Low11"

{regex}: Regular Expressions

Metacharacters:

vals <- c("a.b", "b.c", "c.d")

replace all the “.” with a space

# string, pattern, replace
str_replace(vals, "\\.", " ")
[1] "a b" "b c" "c d"

Default only replaces first:

vals <- c("a.b.c", "b.c.d", "c.d.e")
# string, pattern, replace
str_replace(vals, "\\.", " ")
[1] "a b.c" "b c.d" "c d.e"

Replacing all:

# string, pattern, replace
str_replace_all(vals, "\\.", " ")
[1] "a b c" "b c d" "c d e"

Quantifiers:

Example: find the phone numbers:

strings <- c("550-153-7578",
             "banana",
             "435.114.7586",
             "home: 672-442-6739")

Create a regex that finds all strings that contain a phone number

phone <- "([2-9][0-9]{2})[- .]([0-9]{3})[- .]([0-9]{4})"
#first # between 2-9, the next two are between 0-9 ({2} means 2 previous), then a dash or a dot, then 3 between 0-9, another dash and dot, and then 4 numbers between 0-9

str_detect(strings, phone)
[1]  TRUE FALSE  TRUE  TRUE
# subset only the strings with phone numbers
test <- str_subset(strings, phone)
test
[1] "550-153-7578"       "435.114.7586"       "home: 672-442-6739"

Replace all the “.” with “-” and extract only the numbers (leaving the letters behind). Remove any extra white space.

test |>
  str_replace_all("\\.", "-") |>
  str_replace_all("[a-zA-Z]", "") |>
  str_replace_all("\\:", "") |>
  str_replace_all(" ", "")
[1] "550-153-7578" "435-114-7586" "672-442-6739"

Clean it up and add a column for line and chapter:

original_books <- austen_books() |> # get all of Jane Austen's books
  group_by(book) |>
  mutate(line = row_number(), # find every line
         chapter = cumsum(str_detect(text, regex("^chapter [\\divxlc]", # count the chapters (starts with (^) the word chapter followed by any digit or a roman numeral); cumulative sum so they're in order (assumes ch 2 after ch 1 etc)
                                                 ignore_case = TRUE)))) |> # ignore lower or uppercase
  ungroup() # ungroup it so we have a dataframe again

Visualize positive and negative words:

sent_word_counts |>
  filter(n > 150) |> # take only if there are over 150 instances of it
  mutate(n = ifelse(sentiment == "negative", -n, n)) |> # add a column where if the word is negative make the count negative
  mutate(word = reorder(word, n)) |> # sort it so it goes from largest to smallest
  ggplot(aes(x = n, y = word, fill = sentiment)) +
  geom_col() +
  labs(y = "Contribution to sentiment")
