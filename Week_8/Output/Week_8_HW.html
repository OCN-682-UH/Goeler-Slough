Now, making the above chunks into a function:

benthic_plot <- function(benthicdata, labelset, exclude_codes = c("OOQ", "PVC", "UNK", "MOBL")) {
  
  benthic_pal <- c(
  "Coral" = "#C0504D",
  "CCA" = "#F4C2C2",
  "Macroalgae" = "#0F3800",
  "Turf" = "#16A34A",
  "Sand" = "#E8D8B0",
  "Urchin" = "#5E50A1",
  "Other Invertebrates" = "#0096C7",
  "Cyanobacteria" = "#5C1A48"
)
  
  benthicdata %>%
    left_join(labelset, by = "Label.code") %>%
    filter(!Label.code %in% exclude_codes) %>% #filtering out points that are in pre-defined exclude codes: for my purposes, unknown, outside quadrat, or on PVC
    group_by(Name) %>%
    mutate(Total.points = n()) %>% #calculating the total number of points included per quadrat for benthic cover calculations
    group_by(Date, Site, GoPro.ID, Subcategory, Label.code, ID.Name, Total.points) %>%
    summarise(Point.count = n()) %>%
    mutate(Percent.cover = round((Point.count / Total.points) * 100, 2)) %>%
    
    ggplot(aes(x=GoPro.ID, y=Percent.cover, fill=Subcategory))+
    geom_col()+
    scale_fill_manual(values = benthic_pal)+ #changing colors to pre-defined benthic color palette
    facet_wrap(~ Site + Date, scales = "free_x", strip.position = "bottom") +
    labs(x = "Quadrat",
         y = "Percent Cover",
         fill = "Benthic Category",
         title = "Benthic Cover by Quadrat"
         )+
    theme_minimal()
}
